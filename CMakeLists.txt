cmake_minimum_required(VERSION 3.25)
project(cloudiff_v2 VERSION 0.36.0 LANGUAGES CXX)
cmake_policy(SET CMP0167 NEW)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CLOUDIFF_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSanitizer" OFF)
find_package(nlohmann_json 3.2.0 REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(LibArchive REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATS REQUIRED IMPORTED_TARGET libnats)
pkg_check_modules(UUID REQUIRED IMPORTED_TARGET uuid)
add_library(cloudiff_common STATIC
 src/common/node_identity.cpp src/common/job_state.cpp src/common/nats_client.cpp
 src/common/postgres_client.cpp src/common/heartbeat.cpp src/common/telemetry.cpp src/common/job_engine.cpp src/common/reconciliation.cpp)
target_include_directories(cloudiff_common PUBLIC include)
target_link_libraries(cloudiff_common PUBLIC nlohmann_json::nlohmann_json PostgreSQL::PostgreSQL Threads::Threads OpenSSL::Crypto LibArchive::LibArchive PkgConfig::NATS PkgConfig::UUID)
target_compile_definitions(cloudiff_common PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(cloudiff_common PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(CLOUDIFF_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(cloudiff_common PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cloudiff_common PUBLIC -fsanitize=address,undefined)
endif()
foreach(target IN ITEMS cloudiff-agent cloudiff-control cloudiff-worker)
 string(REPLACE "cloudiff-" "" role ${target})
 add_executable(${target} src/${role}/main.cpp)
 target_link_libraries(${target} PRIVATE cloudiff_common)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endforeach()
target_sources(cloudiff-agent PRIVATE src/agent/npm_publisher_provider.cpp src/agent/artifact_engine.cpp src/agent/runtime_executor.cpp src/agent/mcp_upload.cpp src/agent/secure_distribution.cpp src/agent/admin_observability.cpp)
target_sources(cloudiff-control PRIVATE src/control/build_broker.cpp)
target_sources(cloudiff-worker PRIVATE src/worker/classic_build_worker.cpp)
enable_testing()
foreach(t IN ITEMS node_identity job_state heartbeat nats_client postgres_client job_engine)
 add_executable(test_${t} tests/test_${t}.cpp)
 target_link_libraries(test_${t} PRIVATE cloudiff_common)
endforeach()
add_test(NAME node_identity COMMAND test_node_identity)
add_test(NAME job_state COMMAND test_job_state)
add_test(NAME heartbeat COMMAND test_heartbeat)
add_executable(test_telemetry tests/test_telemetry.cpp)
target_link_libraries(test_telemetry PRIVATE cloudiff_common)
add_test(NAME telemetry COMMAND test_telemetry)
add_test(NAME job_engine COMMAND test_job_engine)

add_executable(cloudiff-reconcile src/reconcile/main.cpp)
target_link_libraries(cloudiff-reconcile PRIVATE cloudiff_common)
target_compile_options(cloudiff-reconcile PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

add_executable(test_reconciliation tests/test_reconciliation.cpp)
target_link_libraries(test_reconciliation PRIVATE cloudiff_common)
target_compile_options(test_reconciliation PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME reconciliation COMMAND test_reconciliation)
